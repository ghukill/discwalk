# Baskets and One Playable Hole — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a disc golf basket that catches discs through a swept chain volume with a bounded momentum budget, and one playable hole with a tee and stroke counting.

**Architecture:** The basket's fat parts (pole, tray) are real Jolt `StaticBody3D` colliders; its thin parts (chains, top band) are swept each physics tick, mirroring how `disc.gd` already marches through tree voxels because leaves can't be physics bodies. `disc.gd` gains ~10 lines that hand its swept segment to `basket.sweep()` and take back a velocity — all chain maths lives in `basket.gd`. `hole.gd` sits on top and knows only about tees and strokes.

**Tech Stack:** Godot 4.7.2 standard, plain GDScript, Jolt Physics. No plugins, no C#, no new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-11-baskets-design.md`

## Global Constraints

- Godot **4.7.2** standard (not .NET). Binary at `/Applications/Godot.app/Contents/MacOS/Godot`.
- Plain GDScript only. No native plugins, no C#, no new addons — the project must keep running unchanged on Linux, macOS and Windows.
- Tabs for indentation, matching every existing script.
- Doc comments use `##`, following the house style in `scripts/*.gd`.
- Tests are headless `SceneTree` scripts in `tools/` that `print(...)` a line per case and end with `print("<NAME> PASS"|"<NAME> FAIL")` then `quit(0|1)`. There is no unit test framework; this is the project's test idiom and the plan follows it.
- All physical sizes in metres. Basket geometry is **independent of `block_size`**.
- Never edit `docs/superpowers/specs/2026-10-11-baskets-design.md` to match the code; if the code must diverge, say so in the task's commit message.

### Deviation from the spec, decided before implementation

The spec says pole, tray and band are "voxel-built through the `voxel_mesh.gd` helpers". **Do not do this.** The tray is 0.635 m across, which at the default 0.25 m `block_size` is 2.5 blocks — voxelising it collapses the tray into a lumpy 3×3 pad and the pole into one column of cubes. Build the basket from Godot primitive meshes (`CylinderMesh`, `TorusMesh`, `BoxMesh`) instead. A manufactured steel basket reading as smooth against a voxel world is correct, not a compromise. `voxel_mesh.gd` is therefore **not** a dependency of `basket.gd`.

- [ ] **Step 3: Create `scripts/basket.gd`**

```gdscript
extends Node3D
## A disc golf basket: pole, tray, top band and chains.
##
## What touches what:
## - Pole and tray are ordinary collision shapes, so a disc clangs off the
##   post and clatters out of the tray on its own.
## - The chains and the top band are NOT physics bodies. They are too thin:
##   a 0.22 m disc at 25 m/s would tunnel straight through. Instead, each
##   physics tick disc.gd hands its swept path to sweep(), which marches it
##   through the chain volume and drains speed. Same trick flora.gd uses for
##   leaves, and for the same reason.
## - The chains absorb a BOUNDED IMPULSE, not a fixed fraction. That is the
##   whole design: a fraction would take the same share of speed at any
##   speed, so tuning it to stop a putt would also stop a drive and every
##   shot would go in. A momentum budget stops a putt dead and lets a drive
##   punch through, which is what a real basket does.
##
## Sizes are PDGA-ish and in METRES, independent of terrain.block_size. The
## basket is built from smooth primitives rather than voxels on purpose: at a
## 0.25 m block size the 0.635 m tray would be two and a half blocks across,
## which destroys the shape. Manufactured steel reading smooth against a grown
## voxel world is the right look anyway.

signal caught(disc)

const BAND_Y := 1.32            ## Top band height (m).
const TRAY_RIM_Y := 0.66        ## Tray rim height (m).
const TRAY_FLOOR_Y := 0.49      ## Tray floor height (m).
const TARGET_RADIUS := 0.3175   ## Tray / band radius (m): 0.635 m across.
const CHAIN_RADIUS := 0.30      ## Chain curtain radius (m).
const CHAIN_TOP := 1.30         ## Chains hang from just under the band (m).
const CHAIN_BOTTOM := 0.75      ## ...down to just over the tray rim (m).
const POLE_RADIUS := 0.05
const CHAIN_COUNT := 12

const COLOR_STEEL := Color(0.62, 0.64, 0.66)
const COLOR_CHAIN := Color(0.78, 0.80, 0.82)
const COLOR_BAND := Color(0.80, 0.62, 0.22)   ## A warm band, easy to pick out.

var terrain: Node3D
var holed: Node = null          ## The disc parked in the tray, or null.
var beam_on := true             ## Tall finder beam, on the P toggle.

var _beacon: MeshInstance3D


func _ready() -> void:
	_build_visual()
	_build_colliders()


## Stand the basket on the ground under `at` (x/z used, y from the terrain).
func place(at: Vector3) -> void:
	var y := at.y
	if terrain != null:
		y = terrain.height_at(at.x, at.z)
	global_position = Vector3(at.x, y, at.z)


func set_beam(on: bool) -> void:
	beam_on = on
	if _beacon != null:
		_beacon.visible = on


# --- building ----------------------------------------------------------------

func _build_visual() -> void:
	var steel := StandardMaterial3D.new()
	steel.albedo_color = COLOR_STEEL
	steel.metallic = 0.6
	steel.roughness = 0.45

	var pole := MeshInstance3D.new()
	var pm := CylinderMesh.new()
	pm.top_radius = POLE_RADIUS
	pm.bottom_radius = POLE_RADIUS
	pm.height = BAND_Y
	pm.radial_segments = 10
	pm.material = steel
	pole.mesh = pm
	pole.position.y = BAND_Y * 0.5
	add_child(pole)

	var band_mat := StandardMaterial3D.new()
	band_mat.albedo_color = COLOR_BAND
	band_mat.metallic = 0.5
	band_mat.roughness = 0.4
	var band := MeshInstance3D.new()
	var bm := TorusMesh.new()
	bm.inner_radius = TARGET_RADIUS - 0.03
	bm.outer_radius = TARGET_RADIUS
	bm.rings = 18
	bm.material = band_mat
	band.mesh = bm
	band.position.y = BAND_Y
	add_child(band)

	# Tray: a shallow open cylinder. A ring of thin slats reads as a cage and
	# costs nothing next to the oaks.
	var tray := MeshInstance3D.new()
	var tm := CylinderMesh.new()
	tm.top_radius = TARGET_RADIUS
	tm.bottom_radius = TARGET_RADIUS * 0.7
	tm.height = TRAY_RIM_Y - TRAY_FLOOR_Y
	tm.radial_segments = 14
	tm.material = steel
	tray.mesh = tm
	tray.position.y = (TRAY_RIM_Y + TRAY_FLOOR_Y) * 0.5
	add_child(tray)

	var chain_mat := StandardMaterial3D.new()
	chain_mat.albedo_color = COLOR_CHAIN
	chain_mat.metallic = 0.7
	chain_mat.roughness = 0.35
	for i in CHAIN_COUNT:
		var a := TAU * i / CHAIN_COUNT
		var bar := MeshInstance3D.new()
		var cm := BoxMesh.new()
		cm.size = Vector3(0.012, CHAIN_TOP - CHAIN_BOTTOM, 0.012)
		cm.material = chain_mat
		bar.mesh = cm
		bar.position = Vector3(cos(a) * CHAIN_RADIUS * 0.85, 
			(CHAIN_TOP + CHAIN_BOTTOM) * 0.5, sin(a) * CHAIN_RADIUS * 0.85)
		add_child(bar)

	_beacon = MeshInstance3D.new()
	var beam := BoxMesh.new()
	beam.size = Vector3(0.08, 12.0, 0.08)
	var beam_mat := StandardMaterial3D.new()
	beam_mat.albedo_color = Color(COLOR_BAND, 0.30)
	beam_mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	beam_mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	beam.material = beam_mat
	_beacon.mesh = beam
	_beacon.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	_beacon.position.y = 6.0
	_beacon.visible = beam_on
	add_child(_beacon)


## Pole and tray are real static bodies. Chains and band are swept (see sweep).
func _build_colliders() -> void:
	var pole := StaticBody3D.new()
	pole.name = "Pole"
	var ps := CollisionShape3D.new()
	var pc := CylinderShape3D.new()
	pc.radius = POLE_RADIUS
	pc.height = BAND_Y
	ps.shape = pc
	ps.position.y = BAND_Y * 0.5
	pole.add_child(ps)
	add_child(pole)

	var tray := StaticBody3D.new()
	tray.name = "Tray"
	var ts := CollisionShape3D.new()
	var tc := CylinderShape3D.new()
	tc.radius = TARGET_RADIUS
	tc.height = TRAY_RIM_Y - TRAY_FLOOR_Y
	ts.shape = tc
	ts.position.y = (TRAY_RIM_Y + TRAY_FLOOR_Y) * 0.5
	tray.add_child(ts)
	add_child(tray)
```

- [ ] **Step 3: Add the knobs and `sweep()` to `basket.gd`**

Add after the `const` block:

```gdscript
## Fraction of speed the chains keep per metre of curtain. Deliberately
## brutal: the ratio is not the discriminator, the impulse cap below is.
@export var chain_drag_per_metre := 0.01
## The most momentum (N·s) the chains can take out of one pass. This is the
## knob that separates a putt from a drive. At a disc's 0.175 kg it is worth
## about 8.6 m/s.
@export var chain_max_impulse := 1.5
## Random sideways rattle, as a fraction of the speed absorbed.
@export var chain_kick := 0.08
## Share of absorbed speed redirected downward, so a dying disc drops in.
@export var chain_down_bias := 0.35
## Speed kept along the hit axis when the top band is clipped.
@export var band_bounce := 0.35
## At or under this speed, inside the cage, the disc is caught (m/s).
@export var catch_speed := 2.5
## March resolution through the chain volume (m).
@export var sweep_step := 0.05
```

Add `var _rng := RandomNumberGenerator.new()` beside the other vars, and `_rng.randomize()` at the end of `_ready()`.

Then add:

```gdscript
# --- chains ------------------------------------------------------------------

## Resolve one physics tick's swept path p0 -> p1 for a disc of `mass` kg.
## Returns {"v": Vector3, "caught": bool, "hit": String} where hit is one of
## "", "chains", "band", "cage". Cheap to call for discs nowhere near us.
func sweep(p0: Vector3, p1: Vector3, v: Vector3, mass: float) -> Dictionary:
	var out := {"v": v, "caught": false, "hit": ""}
	var c := global_position

	# Broad phase: a cylinder a little bigger than the basket, swept by how
	# far we travel this tick. Discs out in the world cost two dot products.
	var seg := p1 - p0
	var dist := seg.length()
	if dist < 1e-6:
		return out
	var mid := (p0 + p1) * 0.5
	var reach := dist * 0.5 + CHAIN_RADIUS + 0.2
	if Vector2(mid.x - c.x, mid.z - c.z).length_squared() > reach * reach:
		return out
	if maxf(p0.y, p1.y) < c.y + TRAY_FLOOR_Y - 0.2 \
			or minf(p0.y, p1.y) > c.y + BAND_Y + 0.2:
		return out

	var vv := v
	var steps := maxi(1, ceili(dist / sweep_step))
	var step_m := dist / steps
	var chain_metres := 0.0
	var prev := p0
	for i in range(1, steps + 1):
		var q := p0 + seg * (float(i) / steps)
		var local := q - c
		var r := Vector2(local.x, local.z).length()
		# Top band: a thin ring. Crossing its plane near the rim clangs.
		if r <= TARGET_RADIUS + 0.04 and r >= TARGET_RADIUS - 0.10:
			if (prev.y - c.y - BAND_Y) * (local.y - BAND_Y) <= 0.0:
				vv.y = -vv.y * band_bounce
				vv.x *= 0.6
				vv.z *= 0.6
				out.v = vv
				out.hit = "band"
				return out
		if r <= CHAIN_RADIUS and local.y >= CHAIN_BOTTOM and local.y <= CHAIN_TOP:
			chain_metres += step_m
		prev = q

	if chain_metres > 0.0:
		out.hit = "chains"
		var speed := vv.length()
		# A fraction would scale with speed and so could never tell a putt
		# from a drive. The cap is what does the work.
		var dv := minf(speed * (1.0 - pow(chain_drag_per_metre, chain_metres)),
			chain_max_impulse / maxf(mass, 0.001))
		dv = minf(dv, speed)
		if speed > 1e-5:
			vv -= vv.normalized() * dv
		vv += Vector3(_rng.randf_range(-1.0, 1.0), 0.0,
			_rng.randf_range(-1.0, 1.0)) * dv * chain_kick
		vv.y -= dv * chain_down_bias

	# Catch test. Runs whether or not chains were touched, and reaches down to
	# the tray floor, so a disc lobbed straight in counts and so does one the
	# chains have already killed on its way down.
	var end_local := p1 - c
	var end_r := Vector2(end_local.x, end_local.z).length()
	if end_r <= CHAIN_RADIUS and end_local.y >= TRAY_FLOOR_Y \
			and end_local.y <= BAND_Y and vv.length() <= catch_speed:
		out.caught = true
		out.hit = "cage"
	out.v = vv
	return out


## Park a caught disc in the tray and hold it there.
func capture(disc: Node) -> void:
	if holed == disc:
		return
	holed = disc
	disc.in_basket = true
	disc.set_beam(false)
	disc.global_position = global_position + Vector3(0, TRAY_FLOOR_Y + 0.10, 0)
	disc.linear_velocity = Vector3.ZERO
	disc.angular_velocity = Vector3.ZERO
	disc.freeze = true
	caught.emit(disc)


## Let go of whatever is in the tray (collect, new world, a fresh hole).
func release() -> void:
	if holed != null and is_instance_valid(holed):
		holed.in_basket = false
		holed.freeze = false
	holed = null
```

```gdscript
extends Node
## One playable hole: a tee, a basket, and a stroke count.
##
## This script knows the rules and nothing about physics. basket.gd knows
## about chains and nothing about strokes. Delete this file and you still
## have a basket to throw at.
##
## Rules, deliberately minimal (docs/superpowers/specs/2026-10-11-baskets-design.md):
## - every disc throw while the hole is unfinished is one stroke
## - cube throws (right click) count for nothing
## - no penalties, no out of bounds: a disc in a lake is just a bad lie
## - holing out completes the hole; later throws do not count
##
## Known and accepted: nothing stops you walking up to the basket and
## dropping a disc in for a 1. Enforcing a tee throw is rule surface this
## version does not want.

signal holed_out(strokes)

const DIST_MIN := 60.0            ## Tee to basket, metres.
const DIST_MAX := 120.0
const TRUNK_CLEAR := 2.0          ## Keep the basket this far off a trunk.
const CANDIDATES := 400

var strokes := 0
var complete := false
var tee := Vector3.ZERO
var basket: Node3D


## Pick a basket spot for `terrain` from its seed. Always returns somewhere:
## a world with a mediocre hole beats a world with none.
static func site(terrain: Node3D) -> Vector3:
	var rng := RandomNumberGenerator.new()
	rng.seed = hash(terrain.world_seed * 104729 + 31)
	var tee_at: Vector3 = terrain.spawn_point()
	var edge: float = terrain.moraine_width + 8.0
	var best := Vector3.ZERO
	var best_score := -1e9
	for i in CANDIDATES:
		var a := rng.randf() * TAU
		var d := rng.randf_range(DIST_MIN, DIST_MAX)
		var x := tee_at.x + cos(a) * d
		var z := tee_at.z + sin(a) * d
		if x < edge or z < edge \
				or x > terrain.world_size - edge or z > terrain.world_size - edge:
			continue
		var score := _score(terrain, x, z, tee_at, d)
		if score > best_score:
			best_score = score
			best = Vector3(x, terrain.height_at(x, z), z)
	if best_score <= -1e8:
		# Nothing passed. Take the least bad spot straight up the middle.
		var fx: float = clampf(tee_at.x, edge, terrain.world_size - edge)
		var fz: float = clampf(tee_at.z - 70.0, edge, terrain.world_size - edge)
		return Vector3(fx, terrain.height_at(fx, fz), fz)
	return best


## Higher is better; -1e9 means illegal.
static func _score(terrain: Node3D, x: float, z: float, tee_at: Vector3,
		d: float) -> float:
	var ix: int = terrain.col_of(x)
	var iz: int = terrain.col_of(z)
	if terrain.is_underwater(ix, iz) or terrain.is_shore(ix, iz):
		return -1e9
	if terrain.lake_edge_distance(x, z) < 2.0:
		return -1e9
	# Flat enough to stand a basket on: no more than one block of fall to any
	# of the four neighbouring columns.
	var h: int = terrain.col_height(ix, iz)
	for o: Vector2i in [Vector2i(1, 0), Vector2i(-1, 0), Vector2i(0, 1), Vector2i(0, -1)]:
		if absi(terrain.col_height(ix + o.x, iz + o.y) - h) > 1:
			return -1e9
	var bs: float = terrain.block_size
	var oaks := 0
	for tr: Dictionary in terrain.flora.trees:
		var tx: float = (tr.pos.x + 0.5) * bs
		var tz: float = (tr.pos.y + 0.5) * bs
		var gap := Vector2(tx - x, tz - z).length()
		if gap < TRUNK_CLEAR + tr.crown * 0.25:
			return -1e9               # basket inside a trunk or its skirt
		# Oaks roughly between tee and basket make the shot interesting.
		if gap < d and Vector2(tx - tee_at.x, tz - tee_at.z).length() < d:
			oaks += 1
	# Prefer mid-range distance, and a handful of oaks in the way.
	var mid := 1.0 - absf(d - (DIST_MIN + DIST_MAX) * 0.5) / ((DIST_MAX - DIST_MIN) * 0.5)
	return mid * 10.0 + minf(oaks, 12) * 1.5


## Begin a hole: tee here, basket there, nothing scored yet.
func start(tee_at: Vector3, basket_at: Vector3) -> void:
	tee = tee_at
	strokes = 0
	complete = false
	if basket != null:
		basket.release()
		basket.place(basket_at)


func count_stroke() -> void:
	if complete:
		return
	strokes += 1


func distance_from(p: Vector3) -> float:
	if basket == null:
		return 0.0
	return Vector2(p.x - basket.global_position.x, p.z - basket.global_position.z).length()
```
